<script type="text/x-dc" data-dc-script>
class Component extends DCLogic {
  state = { step: 1, price: 28, capacity: 550, published: false, publishing: false };
  fmt(n) { return '$' + n.toFixed(2); }
  componentDidMount() {
    this._onI18n = () => this.forceUpdate();
    window.addEventListener('ihype-i18n-change', this._onI18n);
  }
  componentWillUnmount() {
    if (this._onI18n) window.removeEventListener('ihype-i18n-change', this._onI18n);
  }
  renderVals() {
    const t = window.ihypeT || function(k,d){return d};
    const { step, price, capacity, published, publishing } = this.state;
    return {
      steps: [[t('ecStepDetails','Details'), 1], [t('ecStepTickets','Tickets'), 2], [t('ecStepReview','Review'), 3]].map(([label, n]) => ({
        label, active: step >= n, inactive: step < n,
        go: () => { if (n <= step) this.setState({ step: n }); },
      })),
      is1: step === 1, is2: step === 2, is3: step === 3,
      next: () => this.setState({ step: step + 1 }),
      price, capacity,
      priceLabel: this.fmt(price),
      setPrice: (e) => this.setState({ price: +e.target.value || 0 }),
      setCapacity: (e) => this.setState({ capacity: +e.target.value || 0 }),
      artistCut: this.fmt(price * .7), venueCut: this.fmt(price * .2), promoCut: this.fmt(price * .1),
      gross: '$' + (price * capacity).toLocaleString(),
      published, notPublished: !published, publishing,
      publishLabel: publishing ? t('ecLockingCharter','Locking charter…') : t('ecPublishLock','Publish & lock charter 🔒'),
      publish: () => { this.setState({ publishing: true }); setTimeout(() => this.setState({ published: true, publishing: false }), 700); },
    };
  }
}
