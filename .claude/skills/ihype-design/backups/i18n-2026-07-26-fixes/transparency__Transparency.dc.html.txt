class Component extends DCLogic {
  state = { p: 0, ticketBump: 0, payoutBump: 0, secondsAgo: 0 };
  componentDidMount() {
    const start = performance.now(), dur = 1200;
    const tick = (now) => {
      const t = Math.min(1, (now - start) / dur);
      const eased = 1 - Math.pow(1 - t, 3);
      this.setState({ p: eased });
      if (t < 1) this._raf = requestAnimationFrame(tick);
    };
    this._raf = requestAnimationFrame(tick);
    this._live = setInterval(() => {
      this.setState(s => ({ ticketBump: s.ticketBump + Math.ceil(Math.random() * 3), payoutBump: s.payoutBump + Math.round(Math.random() * 84 * 100) / 100 }));
    }, 4000);
    this._clock = setInterval(() => this.setState(s => ({ secondsAgo: s.secondsAgo + 1 })), 1000);
  }
  componentWillUnmount() { if (this._raf) cancelAnimationFrame(this._raf); clearInterval(this._live); clearInterval(this._clock); }
  renderVals() {
    const p = this.state.p;
    const count = (target, prefix) => prefix + Math.round(target * p).toLocaleString();
    const s = this.state.secondsAgo;
    return {
      lastUpdated: s < 5 ? 'Updated just now' : s < 60 ? `Updated ${s}s ago` : `Updated ${Math.floor(s / 60)}m ago`,
      stats: [
        { label: window.ihypeT?window.ihypeT('trnStatArtists'):'Artists on platform', val: count(1247, ''), isAccent: true },
