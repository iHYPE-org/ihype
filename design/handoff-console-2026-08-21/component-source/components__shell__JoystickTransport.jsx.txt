const _JT = { fm: "'JetBrains Mono',monospace" };

/**
 * The right-hand transport — one brass 4-way joystick standing in for five
 * separate buttons (prev / play / pause / next / expand / collapse). A tap
 * plays or pauses; a drag reads as a direction the instant it crosses
 * `THRESH` and the nub snaps back the moment the pointer releases, so the
 * gesture is a flick, not a held position.
 *
 * Left/right mirror `PlayerPill`'s ScrubPlay (drag left = previous, drag
 * right = next) — same physical vocabulary, now standalone brass hardware
 * instead of embedded in the pill. Up opens the full-screen player; down
 * retires it. Whichever axis moves further wins, so a diagonal flick still
 * resolves to one action.
 */
export function JoystickTransport({
  playing = false, canTogglePlay = true, onTogglePlay, onPrev, onNext, onExpand, onCollapse, size = 88,
}) {
  const from = React.useRef(null);
  /* The RAW delta, kept separately from the nub's clamped travel. The gesture
     was being measured from the clamped value, and the clamp (0.22 x size) is
     smaller than the threshold it was compared against (0.26 x size) — so no
     drag could ever cross it and every gesture, however long, resolved as a
     tap. Visual travel and gesture distance are two different quantities. */
  const raw = React.useRef({ x: 0, y: 0 });
  const [d, setD] = React.useState({ x: 0, y: 0 });
  /* THROW is how far the nub travels; THRESH is how far the FINGER must travel
     to read as a direction. They are independent by design — the nub bottoms out
     early so the control feels mechanical, while the gesture keeps measuring. */
  const THROW = size * 0.22;
  const THRESH = size * 0.16;

  const end = () => {
    if (!from.current) return;
    const { x, y } = raw.current;
    from.current = null; raw.current = { x: 0, y: 0 }; setD({ x: 0, y: 0 });
    if (Math.abs(x) < THRESH && Math.abs(y) < THRESH) { onTogglePlay && canTogglePlay && onTogglePlay(); return; }
    if (Math.abs(x) >= Math.abs(y)) { x > 0 ? onNext && onNext() : onPrev && onPrev(); }
    else { y < 0 ? onExpand && onExpand() : onCollapse && onCollapse(); }
  };

  return React.createElement('div', {
    style: { position: 'relative', width: size, height: size, flex: '0 0 auto', perspective: size * 2 },
  },
    React.createElement('span', {
      'aria-hidden': 'true',
      style: {
        position: 'absolute', inset: 0, borderRadius: '9999px',
        border: '2px solid var(--brass-deep)',
        background: 'radial-gradient(circle at 36% 28%, var(--brass) 0%, #7a5a22 62%, #4a3512 100%)',
        boxShadow: '0 6px 14px -5px rgba(0,0,0,.7), inset 0 1px 0 rgba(255,244,214,.5)',
      },
    }),
    React.createElement('span', {
      'aria-hidden': 'true',
      style: {
        position: 'absolute', inset: 1, borderRadius: '9999px', opacity: 0.55,
        background: 'repeating-conic-gradient(from 0deg, rgba(255,244,214,.5) 0deg 2.2deg, rgba(60,42,14,.5) 2.2deg 4.4deg)',
      },
    }),
    React.createElement('span', {
      'aria-hidden': 'true',
      style: {
        position: 'absolute', inset: 6, borderRadius: '9999px',
        background: 'radial-gradient(circle at 40% 32%, var(--walnut-2), #1c1206)',
        border: '1px solid rgba(255,214,160,.2)',
      },
    }),
    ['\u25c0', '\u25b6', '\u25b2', '\u25bc'].map((g, i) => React.createElement('span', {
      key: i, 'aria-hidden': 'true',
      style: {
        position: 'absolute', color: 'var(--lamp)',
        textShadow: '0 0 5px rgba(255,143,45,.7)', userSelect: 'none', WebkitUserSelect: 'none', pointerEvents: 'none',
        left: i === 0 ? 4 : i === 1 ? undefined : '50%', right: i === 1 ? 4 : undefined,
        top: i === 2 ? 3 : i === 3 ? undefined : '50%', bottom: i === 3 ? 3 : undefined,
        zIndex: 1, fontSize: size * 0.11,
        transform: (i === 0 || i === 1) ? 'translateY(-50%)' : 'translateX(-50%)',
      },
    }, g)),
    React.createElement('button', {
      type: 'button', className: 'walnut-frame',
      'aria-label': playing ? 'Pause. Drag for previous, next, or the full player.' : 'Play. Drag for previous, next, or the full player.',
      onPointerDown: (e) => { from.current = { x: e.clientX, y: e.clientY }; raw.current = { x: 0, y: 0 }; e.currentTarget.setPointerCapture(e.pointerId); },
      onPointerMove: (e) => {
        if (!from.current) return;
        const rx = e.clientX - from.current.x, ry = e.clientY - from.current.y;
        raw.current = { x: rx, y: ry };
        setD({
          x: Math.max(-THROW, Math.min(THROW, rx)),
          y: Math.max(-THROW, Math.min(THROW, ry)),
        });
      },
      onPointerUp: end, onPointerCancel: () => { from.current = null; raw.current = { x: 0, y: 0 }; setD({ x: 0, y: 0 }); },
      style: {
        width: '54%', height: '54%', position: 'absolute', left: '23%', top: '23%',
        borderRadius: '9999px', cursor: 'grab', touchAction: 'none', padding: 0,
        background: 'radial-gradient(circle at 36% 30%, #e0c076, var(--brass) 52%, var(--brass-deep))',
        border: '1.5px solid var(--brass-deep)',
        color: 'var(--walnut-3)',
        display: 'grid', placeItems: 'center', fontSize: size * 0.2, lineHeight: 1,
        transformStyle: 'preserve-3d', userSelect: 'none', WebkitUserSelect: 'none',
        transform: 'translate3d(' + d.x + 'px,' + d.y + 'px,6px) rotateX(' + (-d.y * 1.1) + 'deg) rotateY(' + (d.x * 1.1) + 'deg)',
        boxShadow: '0 ' + (4 + d.y * 0.3) + 'px ' + (9 + Math.hypot(d.x, d.y) * 0.4) + 'px -3px rgba(0,0,0,.65), inset 0 1px 0 rgba(255,248,224,.6)',
        transition: from.current ? 'none' : 'transform 200ms cubic-bezier(.28,1.5,.35,1), box-shadow 200ms ease',
      },
    }, React.createElement('span', { 'aria-hidden': 'true' }, playing ? '\u275a\u275a' : '\u25b6'))
  );
}
